% \grant / \grantE arguments:
%   {title/role}{year}{awarding body}{display value}{numeric value, in £k or €k}[PI]
%   - the display value (#5) is exactly what is typeset, unchanged from before;
%   - the numeric value (#6) is the same figure as a plain decimal number,
%     used only to feed the running totals below (kept separate from the
%     display string so the sum never depends on parsing formatted text);
%   - append [PI] when the role is PI, to also add the amount to the
%     PI-only sub-total;
%   - a starred form \grant*{...}/\grantE*{...} excludes the entry from
%     both totals (e.g. an award that was declined and never received).
% \grantsubtotal prints the running GBP/EUR totals (+ PI sub-totals)
% and resets them, so each block below is summed independently.
% The display value (#5) is automatically bolded whenever the numeric
% value (#6) is >=100 (i.e. above 100k, in either currency) -- no manual
% \textbf{} needed in the individual entries any more.
\NewDocumentCommand{\grant}{s m m m m m O{}}{%
  \item {{\em #2} (#3)}. { \dotfill \pounds {\gBoldIfBig{#6}{#5}}}{\newline Awarding body: #4}. %
  \IfBooleanF{#1}{%
    \gAddGBP{#6}%
    \ifstrequal{#7}{PI}{\gAddGBPpi{#6}}{}%
  }%
}
\NewDocumentCommand{\grantE}{s m m m m m O{}}{%
  \item {\em{#2} (#3)}. { \dotfill \euro {\gBoldIfBig{#6}{#5}}}{\newline Awarding body: #4}. %
  \IfBooleanF{#1}{%
    \gAddEUR{#6}%
    \ifstrequal{#7}{PI}{\gAddEURpi{#6}}{}%
  }%
}
